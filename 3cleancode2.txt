// Logic that executes the spotlight overlay (Uses solid high-contrast Neon Yellow)
function highlightEcosystem(selectedEcoName) {
  removeHighlightLayer();
  
  if (selectedEcoName === 'None (Show Full Map)' || !currentEcosystemMap) {
    return;
  }
  
  var ecoIds = ECOSYSTEM_DICT[selectedEcoName];
  var spotlightMask = currentEcosystemMap.eq(ecoIds[0]);
  for (var k = 1; k < ecoIds.length; k++) {
    spotlightMask = spotlightMask.or(currentEcosystemMap.eq(ecoIds[k]));
  }
  
  var spotlightLayer = ee.Image(1).updateMask(spotlightMask);
  highlightLayerRef = mainMap.addLayer(
    spotlightLayer, 
    {palette: ['#FFFF00']}, // Highlighted in bright Neon Yellow!
    'Highlight: ' + selectedEcoName
  );
}

  var yearInt = parseInt(selectedYear, 10);

// Standardized Map Click function shared across Single and Linked Split map windows
function onMapClick(coords) {
  if (!currentEcosystemMap) return;
  
  var clickPoint = ee.Geometry.Point([coords.lon, coords.lat]);
  
  // Clean old markers and drop synced markers across ALL active viewports (mainMap, Left, Right)
  clearMarkers();
  var markerMain = mainMap.addLayer(clickPoint, {color: '#00FFFF'}, 'Inspection Point');
  var markerLeft = leftMap.addLayer(clickPoint, {color: '#00FFFF'}, 'Inspection Point');
  var markerRight = rightMap.addLayer(clickPoint, {color: '#00FFFF'}, 'Inspection Point');
  activeMarkers.push({map: mainMap, layer: markerMain});
  activeMarkers.push({map: leftMap, layer: markerLeft});
  activeMarkers.push({map: rightMap, layer: markerRight});
  
  inspectorResultsPanel.clear();
  inspectorResultsPanel.add(ui.Label('⚡ Querying 2015–2026 trajectory...', {fontStyle: 'italic', color: '#777'}));
  
  var years = [2015, 2016, 2017, 2018, 2019, 2020, 2021, 2022, 2023, 2024, 2025, 2026];
  var yearBands = years.map(function(y) { return generateEcosystemMap(y).rename('y' + y); });
  var multiYearImage = ee.Image(yearBands);
  
  var values = multiYearImage.reduceRegion({
    reducer: ee.Reducer.first(),
    geometry: clickPoint,
    scale: 30 
  });
  
  values.evaluate(function(result, error) {
    inspectorResultsPanel.clear();
    if (error) {
      inspectorResultsPanel.add(ui.Label('Error querying historical data.', {color: 'red'}));
      return;
    }
    
    if (result) {
      // 1. Map our classes to custom vertical tick objects: {v: Class_ID, f: Class_Name}
      var yTicks = CLASSES_LEGEND.map(function(item) {
        return {v: item.id, f: item.name};
      });
      
      // 2. Build the client-side DataTable array [X-axis (Year), Y-axis (Class ID), Tooltip]
      var dataTable = [['Year', 'Class ID', {role: 'tooltip', type: 'string'}]];
      
      years.forEach(function(y) {
        var classId = result['y' + y];
        var matchedClass = CLASSES_LEGEND.filter(function(item) { return item.id === classId; })[0];
        var className = matchedClass ? matchedClass.name : 'Ocean / No Data';
        
        // Push coordinate nodes to our charting array with custom hover tooltips
        dataTable.push([y, classId, y + ': ' + className]);
      });
      
      // 3. Render the Line Chart directly into the viewport panel
      var chart = ui.Chart(dataTable, 'LineChart', {
        title: 'Single-Pixel Trajectory Over Time',
        hAxis: {
          title: 'Year',
          ticks: [2015, 2016, 2017, 2018, 2019, 2020, 2021, 2022, 2023, 2024, 2025, 2026],
          format: '####', // Prevents calendar years from showing comma separators (e.g. 2,015)
          textStyle: {fontSize: 8}
        },
        vAxis: {
          title: 'Ecosystem Class',
          ticks: yTicks, // Dynamically maps the numeric IDs back to their text titles on screen
          direction: -1,
          textStyle: {fontSize: 7},
          viewWindow: {min: 0.5, max: 26.5} // Centers the data lines inside the viewport window
        },
        lineWidth: 2,
        pointSize: 5,
        colors: ['#1B4F72'], // Classy steel blue matching your app header
        legend: {position: 'none'},
        chartArea: {left: 145, top: 30, right: 10, bottom: 40} // Widened left margin to prevent long label clipping
      });
      
      inspectorResultsPanel.add(chart);
    } else {
      inspectorResultsPanel.add(ui.Label('No data found at location.'));
    }
  });
}

// Helper function to dynamically generate the Loss/Gain dynamics bar chart (Type-Safe & Memory-Safe!)
function generateChangeChart(lossMask, gainMask, selectedEco, containerPanel) {
  containerPanel.clear();
  containerPanel.add(ui.Label('⚡ Calculating dynamics (250m scale)...', {fontStyle: 'italic', color: '#777'}));
  
  var reductionGeometry = ee.FeatureCollection(countryBounds).geometry();
  var lossArea = ee.Image.pixelArea().divide(10000).updateMask(lossMask);
  var gainArea = ee.Image.pixelArea().divide(10000).updateMask(gainMask);
  
  var lossStats = lossArea.reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: reductionGeometry,
    scale: 250,     
    maxPixels: 1e9,
    tileScale: 16   
  });
  
  var gainStats = gainArea.reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: reductionGeometry,
    scale: 250,     
    maxPixels: 1e9,
    tileScale: 16   
  });
  
  var combinedStats = ee.Dictionary({
    'Loss': lossStats.get('area'),
    'Gain': gainStats.get('area')
  });
  
  combinedStats.evaluate(function(result, error) {
    containerPanel.clear();
    if (error || !result) {
      containerPanel.add(ui.Label('Error calculating dynamics statistics.', {color: 'red'}));
      return;
    }
    
    var lossHa = Math.round(result.Loss || 0);
    var gainHa = Math.round(result.Gain || 0);
    
    var dataTable = [
      ['Metric', 'Hectares', {role: 'style'}],
      ['Loss', lossHa, '#FF3131'], // Solid neon red for Loss
      ['Gain', gainHa, '#00FF33']  // Neon Green for Gain
    ];
    
    var options = {
      title: selectedEco.split(' (')[0] + ' Dynamics',
      vAxis: {title: 'Hectares', textStyle: {fontSize: 9}},
      hAxis: {title: '', textStyle: {fontSize: 10}},
      legend: {position: 'none'},
      chartArea: {left: 60, top: 30, width: '70%', height: '70%'}
    };
    
    var chart = ui.Chart(dataTable, 'ColumnChart', options);
    containerPanel.add(chart);
  });
}

// Helper function to generate a chart for general landscape changes (Type-Safe & Memory-Safe!)
function generateGeneralChangeChart(changeMask, yearA, yearB, containerPanel) {
  containerPanel.clear();
  containerPanel.add(ui.Label('⚡ Calculating overall change (250m scale)...', {fontStyle: 'italic', color: '#777'}));
  
  var reductionGeometry = ee.FeatureCollection(countryBounds).geometry();
  var changeArea = ee.Image.pixelArea().divide(10000).updateMask(changeMask);
  
  var changeStats = changeArea.reduceRegion({
    reducer: ee.Reducer.sum(),
    geometry: reductionGeometry,
    scale: 250,     
    maxPixels: 1e9,
    tileScale: 16   
  });
  
  changeStats.evaluate(function(result, error) {
    containerPanel.clear();
    if (error || !result) {
      containerPanel.add(ui.Label('Error calculating change statistics.', {color: 'red'}));
      return;
    }
    
    var changeHa = Math.round(result.area || 0);
    
    var dataTable = [
      ['Label', 'Hectares'],
      ['Total Changed', changeHa]
    ];
    
    var options = {
      title: 'Overall Transitioned Hectares (' + yearA + '-' + yearB + ')',
      vAxis: {title: 'Hectares', textStyle: {fontSize: 9}},
      hAxis: {title: '', textStyle: {fontSize: 10}},
      legend: {position: 'none'},
      colors: ['#FF5722'], // Neon Orange for general transition
      chartArea: {left: 60, top: 30, width: '70%', height: '70%'}
    };
    
    var chart = ui.Chart(dataTable, 'ColumnChart', options);
    containerPanel.add(chart);
  });
}

function removeChangeLayers() {
  if (lossLayerRef) { rightMap.remove(lossLayerRef); lossLayerRef = null; }
  if (gainLayerRef) { rightMap.remove(gainLayerRef); gainLayerRef = null; }
  if (generalChangeLayerRef) { rightMap.remove(generalChangeLayerRef); generalChangeLayerRef = null; }
  if (mapALayerRef) { leftMap.remove(mapALayerRef); mapALayerRef = null; } 
  if (mapBLayerRef) { rightMap.remove(mapBLayerRef); mapBLayerRef = null; } 
  rightChartPanel.clear(); 
}

function runChangeDetection() {
  var yearA = parseInt(yearADropdown.getValue(), 10);
  var yearB = parseInt(yearBDropdown.getValue(), 10);
  var selectedEco = changeEcoDropdown.getValue();
  
  if (yearA === yearB) {
    changeLoadingLabel.setValue('❌ Error: Baseline and Comparison years must be different.');
    return;
  }
  
  changeLoadingLabel.setValue('⚡ Generating classification maps and performing change algebra...');
  removeChangeLayers();
  clearMarkers();
  
  // Swap standard map canvas for the SplitPanel (Phase 3!) using mainMap
  if (ui.root.widgets().get(1) !== splitPanel) {
    ui.root.widgets().set(1, splitPanel);
  }
  
  // Align Split Map centers to where the user was looking on the single map
  leftMap.setCenter(mainMap.getCenter().coordinates().get(0).getInfo(), mainMap.getCenter().coordinates().get(1).getInfo(), mainMap.getZoom());
  
  leftMap.layers().reset();
  rightMap.layers().reset();
  
  var mapA = generateEcosystemMap(yearA);
  var mapB = generateEcosystemMap(yearB);
  
  // Add Baseline Map A to the Left Map, and Comparison Map B to the Right Map
  mapALayerRef = leftMap.addLayer(mapA, {min: 1, max: 26, palette: palette}, "Baseline Map: " + yearA);
  mapBLayerRef = rightMap.addLayer(mapB, {min: 1, max: 26, palette: palette}, "Comparison Map: " + yearB);
  
  if (selectedEco === 'All Ecosystems (General Change)') {
    var generalChangeMask = mapA.neq(mapB);
    var generalChangeLayer = ee.Image(1).updateMask(generalChangeMask);
    
    generalChangeLayerRef = rightMap.addLayer(
      generalChangeLayer, 
      {palette: ['#FF5722']}, // Render General Landscape change in solid Neon Orange
      'Change: ' + yearA + ' vs ' + yearB
    );
    
    generateGeneralChangeChart(generalChangeMask, yearA, yearB, rightChartPanel);
    
  } else {
    var ecoIds = ECOSYSTEM_DICT[selectedEco];
    
    var maskA = mapA.eq(ecoIds[0]);
    for (var i = 1; i < ecoIds.length; i++) {
      maskA = maskA.or(mapA.eq(ecoIds[i]));
    }
    
    var maskB = mapB.eq(ecoIds[0]);
    for (var j = 1; j < ecoIds.length; j++) {
      maskB = maskB.or(mapB.eq(ecoIds[j]));
    }
    
    var lossMask = maskA.and(maskB.not());
    var lossLayer = ee.Image(1).updateMask(lossMask);
    
    var gainMask = maskA.not().and(maskB);
    var gainLayer = ee.Image(1).updateMask(gainMask);
    
    // Add Loss (Red) and Gain (Neon Green) on the Right Map for swiping contrast
    lossLayerRef = rightMap.addLayer(lossLayer, {palette: ['#FF3131']}, 'Loss of ' + selectedEco);
    gainLayerRef = rightMap.addLayer(gainLayer, {palette: ['#00FF33']}, 'Gain of ' + selectedEco);
    
    generateChangeChart(lossMask, gainMask, selectedEco, rightChartPanel);
  }
  
  changeLoadingLabel.setValue('');
}

var runChangeButton = ui.Button({
  label: 'Run Comparison',
  onClick: runChangeDetection,
  style: {width: '100%', margin: '5px 0'}
});
