function maskL8clouds(img) {
  var qa = img.select('QA_PIXEL');
  var mask = qa.bitwiseAnd(1 << 3).eq(0).and(qa.bitwiseAnd(1 << 4).eq(0));
  return img.updateMask(mask);
}

function applyL8Scaling(img) {
  var optical = img.select('SR_B.').multiply(0.0000275).add(-0.2);
  var thermal = img.select('ST_B.*').multiply(0.00341802).add(149.0);
  return img.addBands(optical, null, true).addBands(thermal, null, true);
}

// 3. COLLECTION PROCESSING
// ============================================================================

// Process Landsat 8 (Dry)
var dry_st_b10_median = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterBounds(bounds)
  .filterDate(DATES.dry[0], DATES.dry[1])
  .map(maskL8clouds)
  .map(applyL8Scaling)
  .median()
  .select(['ST_B10']); // Direct rename

// Process Sentinel-1 (Dry SAR)
var sar_dry_angle = ee.ImageCollection('COPERNICUS/S1_GRD')
  .filterBounds(bounds)
  .filterDate(DATES.dry[0], DATES.dry[1])
    .filter(ee.Filter.eq('instrumentMode', 'IW'))
  // 2. Ensure all images have BOTH VV and VH polarizations to keep the collection uniform
  .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VV'))
  .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VH'))
  .median()
  .select(['angle']); // Select and rename in one go

var b5 = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterBounds(bounds)
  .filterDate(DATES.dry[0], DATES.dry[1])
  .map(maskL8clouds)
  .map(applyL8Scaling)
  .median()
  .select(['SR_B5']);

// ============================================================================
// 4. TEXTURE ANALYSIS (GLCM)
// ============================================================================

var stats = glcm_band.reduceRegion({
  reducer: ee.Reducer.minMax(),
  geometry: bounds,
  scale: 100, // Increased scale for faster calculation
  bestEffort: true
});

var quantized = glcm_band
  .unitScale(ee.Number(stats.get('SR_B5_min')), ee.Number(stats.get('SR_B5_max')))
  .multiply(31).floor().toInt();

var g4_B5_savg = quantized.glcmTexture({size: 4, average: true})
  .select(['SR_B5_savg']);
